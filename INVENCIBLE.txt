let answer=Math.floor(Math.random()*20)+1;

/* PUZZLE */

let arr=[1,2,3,4,5,6,7,8,""];

const grid=document.getElementById("grid");

function draw(){

grid.innerHTML="";

arr.forEach((v,i)=>{

let d=document.createElement("div");

d.className="tile";

d.innerHTML=v;

d.onclick=()=>move(i);

if(v=="")d.style.background="rgba(255,255,255,.3)";

grid.appendChild(d);

});

}

function move(i){

let empty=arr.indexOf("");

let valid=[empty-1,empty+1,empty-3,empty+3];

if(valid.includes(i)){

[arr[i],arr[empty]]=[arr[empty],arr[i]];

draw();

}

}

function shuffle(){

arr.sort(()=>Math.random()-0.5);

draw();

}

draw();


/* CONFETTI */

const canvas=document.getElementById("confetti");
const ctx=canvas.getContext("2d");

canvas.width=window.innerWidth;
canvas.height=window.innerHeight;

let conf=[];

function startConfetti(){

for(let i=0;i<250;i++){

conf.push({

x:Math.random()*canvas.width,
y:Math.random()*canvas.height-canvas.height,
r:Math.random()*6+3,
c:["#ff69b4","#00bfff","#fff","#ffd700"][Math.floor(Math.random()*4)],
v:Math.random()*4+2

});

}

animate();

}

function animate(){

ctx.clearRect(0,0,canvas.width,canvas.height);

conf.forEach(c=>{

ctx.fillStyle=c.c;

ctx.beginPath();

ctx.arc(c.x,c.y,c.r,0,Math.PI*2);

ctx.fill();

c.y+=c.v;

if(c.y>canvas.height)c.y=-20;

});

requestAnimationFrame(animate);

}

window.onresize=()=>{

canvas.width=window.innerWidth;
canvas.height=window.innerHeight;

};
